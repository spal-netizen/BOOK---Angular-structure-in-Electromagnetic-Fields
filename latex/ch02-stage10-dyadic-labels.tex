% Stage 10 — dyadic labels for Hertzian dipole (displayed in §2.3.1 restore; repeated here for ref closure)
\begin{equation}
\label{eq:ch2.E-from-dyadic}
\E(\mathbf{r})=-\jj\omega\mu_0\int_{\Omega_s}\overline{\mathbf{G}}_{E}(\mathbf{r},\mathbf{r}')\cdot\Jimp(\mathbf{r}')\,d^3r',
\end{equation}
\begin{equation}
\label{eq:ch2.dyadic-free-space}
\overline{\mathbf{G}}_{E}(\mathbf{r},\mathbf{r}')
=\Big(\mathbf{I}+\frac{1}{k_0^2}\grad\grad-\frac{\jj}{k_0}\hat{\mathbf{R}}\times\Big)\,
\frac{e^{-\jj k_0 R}}{4\pi R},
\qquad
\hat{\mathbf{R}}=\frac{\mathbf{r}-\mathbf{r}'}{R},
\end{equation}
for $R=|\mathbf{r}-\mathbf{r}'|>0$ and $k_0=\omega\sqrt{\mu_0\varepsilon_0}$ \cite{stratton1941,jackson1999}.
